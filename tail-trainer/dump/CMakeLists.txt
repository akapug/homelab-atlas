# tail-dump against a llama.cpp checkout, without touching it: cmake -S . -B build -DLLAMA_CPP=/path/to/llama.cpp
cmake_minimum_required(VERSION 3.14)
project(tail-dump CXX C)
set(LLAMA_CPP "" CACHE PATH "a llama.cpp checkout")
if(NOT EXISTS "${LLAMA_CPP}/include/llama.h")
    message(FATAL_ERROR "set -DLLAMA_CPP to a llama.cpp checkout (no include/llama.h under '${LLAMA_CPP}')")
endif()
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
add_subdirectory(${LLAMA_CPP} llama.cpp EXCLUDE_FROM_ALL)
add_executable(tail-dump tail-dump.cpp)
target_link_libraries(tail-dump PRIVATE llama ggml)
target_compile_features(tail-dump PRIVATE cxx_std_17)
